      code: `# Air Canvas - Python

import cv2
import mediapipe as mp
from collections import deque

WIDTH, HEIGHT = 1280, 720

cap = cv2.VideoCapture(0)
cap.set(cv2.CAP_PROP_FRAME_WIDTH, WIDTH)
cap.set(cv2.CAP_PROP_FRAME_HEIGHT, HEIGHT)

mp_hands = mp.solutions.hands
hands = mp_hands.Hands(
    max_num_hands=1,
    min_detection_confidence=0.7,
    min_tracking_confidence=0.7
)

drawer = mp.solutions.drawing_utils
points = deque(maxlen=512)

while True:
    ok, frame = cap.read()

    if not ok:
        break

    frame = cv2.flip(frame, 1)
    rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
    result = hands.process(rgb)

    if result.multi_hand_landmarks:
        hand = result.multi_hand_landmarks[0]

        drawer.draw_landmarks(
            frame,
            hand,
            mp_hands.HAND_CONNECTIONS
        )

        x = int(hand.landmark[8].x * WIDTH)
        y = int(hand.landmark[8].y * HEIGHT)

        points.append((x, y))

    for i in range(1, len(points)):
        cv2.line(
            frame,
            points[i - 1],
            points[i],
            (255, 0, 255),
            5
        )

    cv2.imshow("Air Canvas", frame)

    if cv2.waitKey(1) & 0xFF == 27:
        break

cap.release()
cv2.destroyAllWindows()`

  const codeModal = document.getElementById("codeModal");
  const codeTitle = document.getElementById("codeTitle");
  const codeText = document.getElementById("codeText");
  const closeCode = document.getElementById("closeCode");
  const copyCode = document.getElementById("copyCode");

  document.querySelectorAll("[data-code]").forEach(button => {
    button.addEventListener("click", () => {
      const item = projectCodes[button.dataset.code];

      if (!item) return;

      codeTitle.textContent = item.title;
      codeText.textContent = item.code;

      codeModal.classList.add("open");
      codeModal.setAttribute("aria-hidden", "false");
      document.body.style.overflow = "hidden";
    });
  });

  function closeViewer() {
    codeModal.classList.remove("open");
    codeModal.setAttribute("aria-hidden", "true");
    document.body.style.overflow = "";
  }

  closeCode.addEventListener("click", closeViewer);

  codeModal.addEventListener("click", event => {
    if (event.target === codeModal) {
      closeViewer();
    }
  });

  document.addEventListener("keydown", event => {
    if (event.key === "Escape") {
      closeViewer();
    }
  });

  copyCode.addEventListener("click", async () => {
    try {
      await navigator.clipboard.writeText(codeText.textContent);
      copyCode.textContent = "✓ Copied";

      setTimeout(() => {
        copyCode.textContent = "Copy Code";
      }, 1200);

    } catch (error) {
      copyCode.textContent = "Select & Copy";

      setTimeout(() => {
        copyCode.textContent = "Copy Code";
      }, 1200);
    }
  });
