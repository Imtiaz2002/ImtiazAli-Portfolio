01 / COMPUTER VISION

Air Canvas

Virtual drawing using hand tracking and computer vision.

PythonOpenCVMediaPipe
💻 View Code — Air Canvas
Source Preview
# Air Canvas - Python
import cv2
import mediapipe as mp
from collections import deque

WIDTH, HEIGHT = 1280, 720
cap = cv2.VideoCapture(0)

hands = mp.solutions.hands.Hands(
    max_num_hands=1,
    min_detection_confidence=0.7,
    min_tracking_confidence=0.7
)
drawer = mp.solutions.drawing_utils
points = deque(maxlen=512)

while True:
    ok, frame = cap.read()
    if not ok:
        break

    frame = cv2.flip(frame, 1)
    rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
    result = hands.process(rgb)

    if result.multi_hand_landmarks:
        hand = result.multi_hand_landmarks[0]
        drawer.draw_landmarks(
            frame, hand, mp.solutions.hands.HAND_CONNECTIONS
        )

        x = int(hand.landmark[8].x * WIDTH)
        y = int(hand.landmark[8].y * HEIGHT)
        points.append((x, y))

    for i in range(1, len(points)):
        cv2.line(frame, points[i-1], points[i], (255, 0, 255), 5)

    cv2.imshow("Air Canvas", frame)

    if cv2.waitKey(1) & 0xFF == 27:
        break

cap.release()
cv2.destroyAllWindows()

# Simple Voice Assistant - Python
import webbrowser
import datetime

def assistant(command):
    command = command.lower().strip()

    if "open youtube" in command:
        webbrowser.open("https://www.youtube.com")
        return "Opening YouTube..."

    if "open google" in command:
        webbrowser.open("https://www.google.com")
        return "Opening Google..."

    if "time" in command:
        return datetime.datetime.now().strftime(
            "Current time: %I:%M %p"
        )

    return "Command not recognised yet."

while True:
    command = input("You: ")

    if command.lower() in {"exit", "quit"}:
        print("Assistant: Goodbye!")
        break

    print("Assistant:", assistant(command))
